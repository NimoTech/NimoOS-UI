+# --- md 阵列核查 ------------------------------------------------------------
+
+# md_arrays —— 从 /proc/mdstat 列出所有 md 设备名,每行一个。
+# mdstat 的阵列行形如 "md127 : active raid1 sdb[1] sda[0]",但也可能是
+# "md1 : inactive sdg[1](S)"(未组装/降级,状态与成员之间没有 raid level
+# 字段)或 "md127 : active (auto-read-only) raid1 ..."。正则只锚定到冒号
+# 为止,不要求出现 "active" 或具体 level,三种形状都能匹配 —— 对照
+# NimoOS-LocalStorage 自己的 pkg/mdadm/parse.go,它专门为 inactive 形态
+# 写了第二条正则,注释明说"不加这条,这类阵列会从解析结果里静默消失,
+# 它占着的盘也会被误判为空闲"。这里用单一宽松正则一次性覆盖,不需要
+# 后端那样分两条规则。
+md_arrays() {
+	[[ -r "$RAIDLAB_PROC_MDSTAT" ]] || return 0
+	awk '/^md[0-9]+ +:/ { print $1 }' "$RAIDLAB_PROC_MDSTAT"
+}
+
+# md_members <md> —— 打印成员盘裸设备名,每行一个,字典序。
+# 读 sysfs 的 <md>/md/dev-<name> 目录名,而不是 mdadm --detail:down 要在
+# 停阵列「之前」核查成员,而 mdadm --detail 需要 root;sysfs 普通用户可读。
+#
+# 成员盘处于 faulty 态(mdadm --fail 打过、但尚未 --remove)时,dev-<name>
+# 目录依然在场——内核只是把它 sysfs 下的 state 文件内容改成 faulty,并不
+# 删目录;真正消失只发生在 --remove 之后。Task 6 的验收流程会用 --fail
+# 人工制造故障盘来验证 down 的行为,这意味着该盘依然会被这里认作成员:
+# 这是符合预期的——它没有被踢出阵列,仍应纳入"是否全假盘"的核查。
+md_members() {
+	local md="${1:-}"
+	[[ -n "$md" ]] || return 0
+	# 路径穿越防护:目前唯一调用方是 md_arrays() 的输出(保证形如
+	# md[0-9]+),但这里是"停阵列 / 改 fstab"之前唯一的核查关卡,不能假设
+	# 调用方永远守规矩——同 is_fake_disk 那道护栏一样的教训。含 / 或恰为
+	# ./.. 的值不允许拼进 sysfs 路径,以免 dir 逃出 RAIDLAB_SYSFS_ROOT。
+	case "$md" in
+	*[!A-Za-z0-9._-]*|.|..)
+		return 0
+		;;
+	esac
+	local dir="$RAIDLAB_SYSFS_ROOT/block/$md/md"
+	[[ -d "$dir" ]] || return 0
+	local path base
+	for path in "$dir"/dev-*; do
+		[[ -e "$path" ]] || continue
+		base="$(basename "$path")"
+		echo "${base#dev-}"
+	done | sort
+}
+
+# assert_md_all_fake <md> —— 阵列所有成员都是假盘才返回 0。
+# 无成员也算失败:那意味着我们读不到成员信息,不能据此做破坏性操作。
+# 这一支同时覆盖两种情况——阵列确实存在但 sysfs 下读不到任何 dev-*
+# (异常态),以及传入的 md 根本不对应任何阵列(拼写错误 / 阵列已停)。
+# 两种情况都判失败是刻意的保守选择:调用方(Task 5 的 cmd_down)只会拿
+# md_arrays() 报告「确实在场」的阵列来调用这个函数,所以这里选"读不到
+# 就拒绝",而不是"读不到就当作没有非假盘、放行"。
+assert_md_all_fake() {
+	local md="${1:-}"
+	local members
+	members="$(md_members "$md")"
+	if [[ -z "$members" ]]; then
+		echo "raidlab: 阵列 '$md' 读不到任何成员盘,拒绝对它做破坏性操作。" >&2
+		return 1
+	fi
+	local dev
+	while IFS= read -r dev; do
+		[[ -n "$dev" ]] || continue
+		if ! is_fake_disk "$dev"; then
+			echo "raidlab: 阵列 '$md' 的成员 '$dev' 不是假盘,拒绝操作该阵列。" >&2
+			echo "         这不是测试台建的阵列,可能承载真实数据。" >&2
+			return 1
+		fi
+	done <<< "$members"
+	return 0
+}

+#     前缀正则 ^\/dev\/md 会连 /dev/md0p1、/dev/mdanything 这类用户自己
+#     写的条目一起删掉,是不必要的误伤面。
+#   - service/snapshot/mount.go 的 SnapshotsDir = filepath.Join(挂载点,
+#     ".snapshots"),挂载点恒以「路径分隔符 + .snapshots」作为独立的最后
+#     一段。后缀正则 \.snapshots$(不要求前面的 /)会连 "/mnt/foo.snapshots"
+#     这种目录名字面上凑巧以该串结尾、但根本不是这个功能挂载点的条目也
+#     一起删掉。
+fstab_drop_snapshots() {
+	awk '$1 ~ /^\/dev\/md[0-9]+$/ && $2 ~ /\/\.snapshots$/ { next } { print }'
