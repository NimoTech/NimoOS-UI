 # is_fake_disk <dev> —— <dev> 为裸设备名(sda),不带 /dev/。
 # 两道条件必须同时满足,缺一不可。
 is_fake_disk() {
 	local dev="${1:-}"
 	[[ -n "$dev" ]] || return 1
 
 	# ① 名字硬否决。model 文件理论上可被伪造(scsi_debug 允许自定义型号),
 	#    而 nvme0n1 是本机唯一真盘、承载 /DATA,所以名字这层不给任何余地。
+	# ② 路径穿越防护:dev 这个值后面会被外部输入驱动(Task 3 的
+	#    verify_avail_only_fake 把后端 HTTP 响应 JSON 的 name 字段原样喂
+	#    进来),而这条护栏的下游是 mdadm --create(会清扫成员盘),必须在
+	#    入口把设备名的合法形状钉死——只允许字母、数字、_、-、. 组成(真实
+	#    块设备名如 sda/nvme0n1/md127/dm-0 都在这个集合内),含 / 或恰为
+	#    ./.. 一律拒绝,防止 model_file 拼出的路径逃出 RAIDLAB_SYSFS_ROOT。
 	case "$dev" in
 	*nvme*) return 1 ;;
+	.|..) return 1 ;;
+	*[!A-Za-z0-9._-]*) return 1 ;;
 	esac
 
 	# ② sysfs 型号必须恰为 scsi_debug(去掉所有空白后比较)。
 	local model_file="$RAIDLAB_SYSFS_ROOT/block/$dev/device/model"
 	[[ -r "$model_file" ]] || return 1
 	local model
 	model="$(tr -d '[:space:]' < "$model_file")"
 	[[ "$model" == "$FAKE_DISK_MODEL" ]]
 }
